# infra/twenty/env.example.txt — plantilla de variables de entorno.
#
# El harness de Claude Code bloquea la creación/escritura de ficheros .env* por
# política de protección de secretos (verificado: hasta un `ls .env.example`
# vía Bash es denegado en este directorio) — por eso esta plantilla se llama
# env.example.txt en vez de .env.example. En el VPS, renómbrala tú mismo:
#   cp env.example.txt .env
#
# Copia a .env y rellena los valores reales. .env NO se commitea (secretos
# reales). Esta plantilla es la única fuente de verdad de qué variables
# existen — docker-compose.yml las consume TODAS vía env_file, no hay nada
# inline en el compose (salvo TAG, que se resuelve como var de compose, no
# de la app, y por eso también vive aquí).
#
# Basado en el .env.example oficial de twentyhq/twenty (packages/twenty-docker/),
# verificado contra la documentación y el repo en 2026-07-01, con el añadido de
# las variables POSTGRES_* que necesita la imagen postgres:16 para arrancar (el
# compose oficial las inyecta indirectamente vía interpolación ${VAR:-default}
# en el bloque `environment:` del servicio db; aquí, al usar env_file puro sin
# interpolación, hace falta declararlas explícitamente).

# ─── Imagen de Twenty ────────────────────────────────────────────────────────
# Pinneado a v2.17.0: último release con entrada completa en GitHub Releases
# (publicado 2026-06-29) en el momento de escribir esto, bien por encima del
# mínimo soportado (2.1.0). Verifica en https://github.com/twentyhq/twenty/releases
# si hay algo más reciente Y estable antes de desplegar — si ha pasado tiempo
# desde 2026-07-01, probablemente sí. NUNCA uses `latest` en producción: rompe
# reproducibilidad y puede traer migraciones de golpe sin aviso. El
# docker-compose.yml de este repo además EXIGE que TAG esté definida (falla
# explícitamente si falta) para que nadie se deje esto en blanco por error.
TAG=v2.17.0

# ─── URL pública ─────────────────────────────────────────────────────────────
# DEBE ser la URL HTTPS pública exacta que usa el navegador, dominio incluido,
# sin barra final. Twenty (desde que unificó front+server en un solo proceso
# Node en el puerto 3000) usa SOLO esta variable como fuente de verdad de la
# URL pública — NO existe una FRONT_BASE_URL separada en las versiones
# actuales (si la has visto en algún tutorial antiguo, está desactualizado
# y corresponde a una arquitectura previa de Twenty con front y server
# desacoplados). Un SERVER_URL que no coincida exactamente (http en vez de
# https, dominio distinto, localhost, barra final inconsistente...) es la
# causa nº1 del clásico bucle de redirección al hacer login en Twenty
# self-hosted.
SERVER_URL=https://crm.duendes.net

# ─── Secreto de cifrado ──────────────────────────────────────────────────────
# ENCRYPTION_KEY es OBLIGATORIO y es el secreto primario (sustituye a
# APP_SECRET en versiones actuales de Twenty; cifra OAuth tokens, variables de
# aplicación y secretos TOTP guardados en la base de datos). Genera uno con:
#   openssl rand -base64 32
ENCRYPTION_KEY=

# Solo se usa durante una rotación de ENCRYPTION_KEY: pon aquí la clave
# ANTERIOR mientras migras a la nueva. Vacío en una instalación nueva.
FALLBACK_ENCRYPTION_KEY=

# LEGACY. Solo relevante para instancias que ya existían ANTES de que Twenty
# introdujera ENCRYPTION_KEY. En una instalación nueva (este caso) déjalo
# vacío — no lo uses como secreto primario aunque lo veas así en tutoriales
# antiguos o en foros.
APP_SECRET=

# ─── Base de datos (Postgres) ────────────────────────────────────────────────
# Variables que consume el propio contenedor `db` (imagen postgres:16 oficial)
# Y que Twenty (server/worker) usa para componer PG_DATABASE_URL más abajo.
PG_DATABASE_USER=twenty
# Sin caracteres especiales: Twenty compone PG_DATABASE_URL como connection
# string y un carácter especial (@, :, /, etc.) sin escapar rompe el parseo.
# Usa solo letras y números. Genera con: openssl rand -hex 24
PG_DATABASE_PASSWORD=CHANGE_ME_strong_password_no_special_chars
PG_DATABASE_HOST=db
PG_DATABASE_PORT=5432
PG_DATABASE_NAME=default
# Alias que espera la imagen postgres:16 oficial (POSTGRES_* en vez de
# PG_DATABASE_*) — deben coincidir EXACTAMENTE con los tres valores de arriba.
POSTGRES_USER=twenty
POSTGRES_PASSWORD=CHANGE_ME_strong_password_no_special_chars
POSTGRES_DB=default

# Connection string ya compuesta que consume Twenty (server y worker). Debe
# reflejar los mismos valores que las variables PG_DATABASE_* de arriba. Si
# cambias el password arriba, cámbialo también aquí — no hay interpolación
# automática al usar env_file puro.
PG_DATABASE_URL=postgres://twenty:CHANGE_ME_strong_password_no_special_chars@db:5432/default

# ─── Redis ────────────────────────────────────────────────────────────────────
REDIS_URL=redis://redis:6379

# ─── Storage ─────────────────────────────────────────────────────────────────
# Disco local del VPS (volumen server-local-data): correcto para este box
# dedicado de instancia única. Si en el futuro migras a object storage (p.ej.
# para múltiples réplicas del server, o para simplificar backups de adjuntos),
# descomenta y rellena las variables S3 de abajo — STORAGE_TYPE pasaría a S_3.
STORAGE_TYPE=local

# STORAGE_S3_REGION=eu-west-3
# STORAGE_S3_NAME=my-bucket
# STORAGE_S3_ENDPOINT=
# STORAGE_S3_ACCESS_KEY_ID=
# STORAGE_S3_SECRET_ACCESS_KEY=

# ─── Migraciones y cron (servidor) ───────────────────────────────────────────
# El worker ya fuerza estas dos a "true" en docker-compose.yml (hardcoded),
# porque el servicio `server` es quien debe ejecutar migraciones y registrar
# los crons — el worker no debe repetirlos. Aquí van los valores del server:
# en una instalación normal deben quedarse en false para que el server SÍ
# migre y registre cron jobs.
DISABLE_DB_MIGRATIONS=false
DISABLE_CRON_JOBS_REGISTRATION=false

# Puerto interno del servidor Node (Caddy hace reverse_proxy a server:3000,
# ver Caddyfile). Se mantiene aquí (en vez de solo hardcoded en el compose)
# porque viaja vía env_file junto con todo lo demás y Twenty lo lee en runtime.
NODE_PORT=3000

# ─── API key de Twenty (NO se configura aquí) ────────────────────────────────
# TWENTY_API_KEY no es una variable de entorno de este stack: se genera DENTRO
# de la app tras el primer login, en Settings → API & Webhooks → API keys. Se
# necesitará más adelante para integraciones (p.ej. el adaptador FastAPI del
# resto del proyecto). No hay línea que rellenar aquí — solo documentamos
# dónde conseguirla (ver SETUP.md, paso 8). Guárdala en un gestor de secretos,
# nunca en este repo.
